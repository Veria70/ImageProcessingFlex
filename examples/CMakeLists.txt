foreach(example quickstart gallery count_objects benchmark)
    add_executable(ipf_${example} ${example}.cpp)
    target_link_libraries(ipf_${example} PRIVATE ipf::ipf)
    if(MSVC)
        target_compile_options(ipf_${example} PRIVATE /W4 /permissive- /utf-8)
    else()
        target_compile_options(ipf_${example} PRIVATE -Wall -Wextra)
    endif()
endforeach()
